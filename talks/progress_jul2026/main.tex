%---------------------------------------------------------------- IDEA
\begin{frame}{The idea: one model for all amplitudes}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \footnotesize
      \begin{itemize}
        \item A surrogate replaces the matrix-element call:
          \begin{center}\fbox{particle momenta $\longrightarrow$ \Amp}\end{center}
        \item Until now: \textbf{one network per process} --- each starts from zero, no physics shared.
        \item Here: a \emph{single} network trained \textbf{jointly} on many processes ($ee\!\to\!WW$, $t\bar t$, $u\bar u(g)$, $\gamma\gamma$, \dots up to \textbf{448 datasets}), tree \emph{and} loop level.
        \item Then \textbf{fine-tuned} to any new process or order with a fraction of the data.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \includegraphics[width=0.82\linewidth]{amp_surface_pts.png}\\[0.2em]
      {\scriptsize\color{gray} \Amp{} over a phase-space slice; \textcolor{warn}{dots} $=$ training samples.}
    \end{column}
  \end{columns}
  \takeaway{Learn the physics once, reuse it everywhere: new process $=$ cheap fine-tune, not a new model.}
\end{frame}

%---------------------------------------------------------------- MODEL
\begin{frame}{The model in one slide: a Lorentz-equivariant transformer}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \footnotesize
      \begin{itemize}
        \item A \textbf{transformer}: a network for \emph{sets} of objects --- the particles of one event. Each particle mixes information from the others with learned weights (``attention''); particle number and ordering don't matter. Built for events.
        \item \textbf{LLoCa} makes it \emph{exactly} Lorentz-equivariant: a small net predicts a \textbf{local frame} per particle; momenta expressed in their own frames become invariant numbers a plain transformer can digest.
        \item Cheap: $\sim$\textbf{20$\times$ fewer FLOPs} than fully-tensorial equivariant nets (L-GATr) --- and \emph{more accurate} at equal compute or wall-time in our shoot-out (backup).
      \end{itemize}
    \end{column}
    \begin{column}{0.54\textwidth}
      \centering
      \includegraphics[width=\linewidth]{lloca_white.png}\\[0.4em]
      {\scriptsize\color{gray} LLoCa: learn each particle's local frame; any backbone becomes Lorentz-equivariant.}
    \end{column}
  \end{columns}
  \takeaway{Exact Lorentz symmetry at the cost of a plain transformer.}
\end{frame}

%---------------------------------------------------------------- INPUTS
\begin{frame}{What the network is fed (beyond the momenta)}
  \footnotesize
  \begin{itemize}
    \item Each particle $=$ momentum $+$ its \textbf{quantum numbers}
      $[\,Q,\ \mathrm{spin},\ \log m,\ T_3,\ B,\ L,\ \mathrm{color},\ C_2\,]$ --- not a particle ``dictionary'' (next slide).
    \item Plus per-dataset physics the momenta \textbf{cannot} encode, broadcast as extra scalars:
      \begin{itemize}
        \item \textbf{coupling order} $[n_{\rm loops},\,\alpha_s\text{-power}]$ and coupling value $\log\alpha$ $\Rightarrow$ LO and NLO mix in one model;
        \item \textbf{off-shellness} $s_{\rm prop}-M^2$: the event's distance to the pole of an \emph{internal} ($Z$/top/Higgs) propagator.
      \end{itemize}
    \item And a detail that mattered a lot: \Amp{} spans $\sim$13 orders of magnitude $\Rightarrow$ log-compress and standardize \textbf{per process} (a global transform left most processes untrainable; median error $6\times$ worse).
  \end{itemize}
  \takeaway{Guiding principle (verified by A/Bs): a feature helps exactly when it supplies what the external momenta cannot recover.}
\end{frame}

%---------------------------------------------------------------- PID VS QN
\begin{frame}{Why quantum numbers, not particle IDs}
  \footnotesize
  Freeze a pretrained model, predict processes it \textbf{never trained on} --- true vs.\ predicted amplitude distributions:
  \vskip0.2em
  \centering
  \includegraphics[height=0.52\textheight]{zero_shot_histograms.pdf}
  \vskip0.2em
  \raggedright
  \begin{itemize}
    \item With \textbf{quantum numbers} the model produces sensible amplitudes for both held-out processes --- including $ee\!\to\!t\bar t$, whose \textbf{top quark it never saw} in training.
    \item With a \textbf{particle-ID dictionary}, $ee\!\to\!t\bar t$ \emph{cannot even be encoded} --- there is no token for the top.
  \end{itemize}
  \takeaway{Encoding particles by their physics, not their name, is what makes transfer to new processes possible at all.}
\end{frame}

%---------------------------------------------------------------- AB HIGHLIGHTS
\begin{frame}{Feature A/Bs: the highlights}
  \footnotesize
  Every candidate feature was A/B-tested (tuned baselines, compared on the non-regularized validation loss):
  \vskip0.3em
  \centering
  \begin{tabular}{@{}llrrl@{}}
    \toprule
    Feature & test & off & on & verdict \\
    \midrule
    coupling $\log\alpha$ (not inferable from kinematics) & $ee\,u\bar uggg$ & $0.057$ & $0.003$ & \textbf{win 18$\times$} \\
    external mass from momenta & near-threshold $t\bar t$ & $2.1\text{e-}5$ & $1.7\text{e-}5$ & redundant \\
    internal mass as a raw scalar & $M_Z$ scan, $ee\,\mu\mu$ & $0.076$ & $0.080$ & \textcolor{warn}{\textbf{fails}} \\
    \textbf{off-shellness} $s_{\rm prop}-M^2$ & $M_Z$ scan & $0.076$ & $9.4\text{e-}5$ & \textbf{win 810$\times$} \\
    \bottomrule
  \end{tabular}
  \vskip0.5em
  \raggedright
  \begin{itemize}
    \item One pattern: a feature wins \textbf{iff} it supplies what the momenta cannot recover (a coupling, an internal mass) \emph{in a usable form} --- the raw mass scalar fails, the kinematic distance-to-pole succeeds.
  \end{itemize}
\end{frame}

%---------------------------------------------------------------- FLAGSHIP
\begin{frame}{Flagship: off-shellness conditioning, 50$\times$ at scale}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \footnotesize
      \begin{itemize}
        \item Confirmed where it matters: \textbf{44 datasets} spanning the $Z$, top and Higgs resonances, full data, 30k steps.
        \item Best validation loss $0.268 \to 0.0053$: \textbf{50$\times$}.
        \item On by default in all production runs.
      \end{itemize}
    \end{column}
    \begin{column}{0.58\textwidth}
      \centering
      \includegraphics[width=\linewidth]{levers_offshell_ab.pdf}\\[0.1em]
      {\scriptsize\color{gray} validation loss vs.\ training step, feature off vs.\ on.}
    \end{column}
  \end{columns}
  \takeaway{Telling the network how far each event is from the internal poles is the single largest gain found.}
\end{frame}

%---------------------------------------------------------------- SCOREBOARD
\begin{frame}{Everything else tried, in one table}
  \footnotesize
  \centering
  \begin{tabular}{@{}lll@{}}
    \toprule
    What & Evidence & Verdict \\
    \midrule
    per-process preprocessing & $6\times$ lower median error & \win \\
    one-hot spin/color, massless flag & removing color costs $17\%$ at 448 sets & \win \\
    geometric-mean joint loss & keeps every process improving equally & \win \\
    full fine-tune $+$ layer-decay lr & beats LoRA / freezing / EWC everywhere & \win \\
    LLoCa backbone & $3$--$9\times$ better than L-GATr at equal budget & \win \\
    \addlinespace
    diagram-graph encoder & removing it \emph{gains} 35\% at scale & \lose \\
    adaptive ``balanced'' sampler & loses $3\times$ to plain uniform sampling & \lose \\
    MLP particle embedder & not load-bearing at scale & \lose \\
    pairwise attention bias & rescues hard processes, taxes the rest 30--50\% & \lose \\
    \bottomrule
  \end{tabular}
  \vskip0.5em
  \raggedright
  \begin{itemize}
    \item Two of the ``kept'' features looked \emph{bad} in short small-data tests and only proved out at 448 datasets --- and the plausible-on-paper diagram encoder actively hurt. Cheap A/Bs can mislead in both directions.
  \end{itemize}
\end{frame}

%---------------------------------------------------------------- SCALING
\begin{frame}{Result: scaling is a power law set by the physics}
  \begin{columns}[T]
    \begin{column}{0.40\textwidth}
      \footnotesize
      \begin{itemize}
        \item Error vs.\ training compute follows $L= A\,C^{-\alpha}+L_\infty$.
        \item The exponent \textbf{tracks final-state multiplicity}, following the bound $\alpha=4/\mathrm{DOF}$, $\mathrm{DOF}=3n_f-4$: $\alpha\!\approx\!2$ for $2\!\to\!2$ down to $\approx\!1$ for $2\!\to\!4$.
        \item $\Rightarrow$ the cost of a precision target is \textbf{predictable from the process alone}, before training.
      \end{itemize}
    \end{column}
    \begin{column}{0.60\textwidth}
      \centering
      \includegraphics[width=0.92\linewidth]{alpha_vs_multiplicity_overlay.pdf}\\[0.1em]
      {\scriptsize\color{gray} fitted exponents per process family vs.\ the $4/\mathrm{DOF}$ bound (grey wedge); stars $=$ fine-tuned one-loop targets.}
    \end{column}
  \end{columns}
\end{frame}

%---------------------------------------------------------------- TRANSFER
\begin{frame}{Result: fine-tuning beats from-scratch, and never plateaus}
  \begin{columns}[T]
    \begin{column}{0.40\textwidth}
      \footnotesize
      \begin{itemize}
        \item Targets: \textbf{one-loop} $ee\!\to\!u\bar u$ / $t\bar t$, fine-tuned from jointly-pretrained models vs.\ trained from scratch.
        \item Fine-tuning is \textbf{lower at every budget} and \textbf{keeps descending} where from-scratch flattens: 1--2 orders of magnitude lower floor.
        \item Holds even for a foundation that \textbf{never saw the target process} $\Rightarrow$ genuine transfer, not memorization.
      \end{itemize}
    \end{column}
    \begin{column}{0.60\textwidth}
      \centering
      \includegraphics[width=\linewidth]{compute_scan_eettbar_wt.pdf}\\[0.1em]
      {\scriptsize\color{gray} $ee\!\to\!t\bar t$ one-loop: test error vs.\ wall-time, panels $=$ dataset size $10^3$--$10^6$.}
    \end{column}
  \end{columns}
  \takeaway{Pretraining pays off exactly where it should: little data, new process, higher order.}
\end{frame}

%---------------------------------------------------------------- BIG RUN
\begin{frame}{The 448-process run: bulk learns, a tail resists}
  \footnotesize
  One model on \textbf{448 datasets} ($10^5$ events each). Median per-process loss $2\times10^{-4}$ --- but $50$ datasets fail, the \emph{same ones every time}. Two understood mechanisms:
  \vskip0.3em
  \begin{itemize}
    \item \textbf{Resonance needles}: flat sampling barely populates the poles --- the $Z$ peak is $0.3\%$ of a wide $\sqrt s$ window, worse when the resonance sits in an \emph{internal} invariant mass. The same processes with windows centred on the pole learn fine $\Rightarrow$ a \textbf{data-generation} problem, not a model one.
    \item \textbf{Gradient competition}: the log-space loss weights each process by $1/\mathrm{MSE}$, so a converged process outweighs a stuck one by $\sim\!10^4$ and laggards never gain traction.
  \end{itemize}
  \vskip0.3em
  Next steps: resonance-aware event generation (importance-sample the poles), and flooring the per-process loss weight ($\log(\mathrm{MSE}_p+\tau)$) --- a one-line change with a decisive test identified.
  \takeaway{Model and feature set are settled; the frontier is data generation and loss balancing for the hard tail.}
\end{frame}

%---------------------------------------------------------------- SUMMARY
\begin{frame}{Summary}
  \footnotesize
  \begin{itemize}
    \item A single Lorentz-equivariant transformer learns \textbf{hundreds of processes at once}, tree and loop level.
    \item Particles enter by their \textbf{quantum numbers}; couplings and \textbf{off-shellness} (50$\times$) supply what momenta can't.
    \item Every design choice A/B-tested; the survivors are in the scoreboard, the rest discarded on evidence.
    \item Scaling is \textbf{predictable from the process} ($\alpha\simeq4/\mathrm{DOF}$); fine-tuning transfers to new processes and orders with a 1--2 orders-of-magnitude lower floor.
    \item Remaining frontier: the $\sim$50 resonance-dominated datasets $\to$ smarter event generation $+$ loss balancing.
  \end{itemize}
