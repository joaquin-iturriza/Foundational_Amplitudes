%---------------------------------------------------------------- EMPHASIS 1
\begin{frame}{Emphasis: the coupled $(\mu,\sigma)$ fit fails structurally}
  \vfill
  \begin{center}
    {\large Amplitudes are \textbf{deterministic} $\Rightarrow$ true $\sigma\to0$,\\[0.4em]
     the $\sigma$ gradient $\sigma^{-1}(1-r^2/\sigma^2)$ blows up,\\[0.4em]
     and \texttt{clip\_grad\_norm} is \textbf{global} --- so it throttles $\mu$ too.}\\[1.6em]
    {\footnotesize\color{gray} binds on 97.5\% of steps (median grad-norm 51 vs clip 5)}\\[2.0em]
    {\large\color{good} Fix: two-stage --- train $\mu$ with MSE, freeze, then fit the 258-param $\sigma$ readout.}\\[0.6em]
    {\footnotesize\color{gray} calibrated $\sigma$ at zero accuracy cost, $\sim$1--3 min}
  \end{center}
  \vfill
\end{frame}
