%---------------------------------------------------------------- sampling
\begin{frame}{Default and catalog sampling}
  \vfill
  \begin{minipage}{0.97\textwidth}
    \small
    \begin{itemize}\setlength{\itemsep}{0.8em}
      \item Default, the generator's own sampling. MadEvent, $pp$ at 13 TeV or $e^+e^-$ at fixed energies, events distributed as the cross section
      \item Catalog, our sampling of the partonic process. $\sqrt{\hat s}$ log uniform up to 6 TeV, a bulk part, a forward part and a part flat in $\log|\mathcal M|^2$
      \item Trained on 0.75 catalog plus 0.25 default
    \end{itemize}
  \end{minipage}
  \vfill
\end{frame}
