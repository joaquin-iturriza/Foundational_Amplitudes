\begin{frame}{Per-process statistics cannot zero shot}
  \vfill
  \begin{minipage}{0.97\textwidth}
    \small
    \begin{itemize}\setlength{\itemsep}{0.6em}
      \item $y=(\log|\mathcal M|^2-\mu_p)/\sigma_p$, with $\mu_p$, $\sigma_p$ from the process's own events
      \item $\mu_p$, $\sigma_p$ move with how the events were drawn, sampling, $\sqrt{\hat s}$ window, cuts
      \item A held-out process has none. Predicted from the process, $\mathrm{MSE}=1.27$ on the 43 new, $0.46$ with its true $\mu_p$, $\sigma_p$
      \item Needed, a target and inputs that depend on neither
    \end{itemize}
  \end{minipage}
  \vfill
\end{frame}

\begin{frame}{Targets, inputs and conditioning}
  \vfill
  \begin{minipage}{0.97\textwidth}
    \footnotesize
    \begin{itemize}\setlength{\itemsep}{0.35em}
      \item Shared standardization, $y=(T-\mu)/\sigma$ with one $\mu$, $\sigma$ for all processes, on every $T$ below
      \item $\hat s$ factor, $T=\log\big[|\mathcal M|^2(\hat s/\Lambda_0^2)^{N-4}\big]$, $\Lambda_0^2=289$ GeV$^2$ fitted on the trained processes
      \item $+$ couplings, $|\mathcal M|^2/C_p$, $C_p=\max_{\rm diagrams}\prod_{\rm vertices}(c_v)^2$ (SM couplings, $v$ for dimensionful ones)
      \item Dimensionless, $|\mathcal M|^2\Lambda^{2(N-4)}$, $\Lambda^2=\hat s$. Pair scale, $\log\Lambda^2=\langle\log|2p_i\cdot p_j|\rangle_{i<j}$
      \item $+$ inputs, $p_i/\sqrt{\hat s}$ and $\log_{10}(m_i/\sqrt{\hat s})$, no other input in GeV
      \item $+$ coupling orders, $|\mathcal M|^2/\big[(4\pi\alpha_s)^{q}(4\pi\alpha)^{e}\big]$, $q$, $e$ the QCD and QED orders
      \item Emission scales, $\Lambda^{2(N-4)}\to\prod_{k=1}^{N-4}q_k^2$, $q_k^2=\min_{a,b}\frac{s_{ak}s_{bk}}{s_{ab}}+(m_k)^2$, softest particle removed first
      \item Diagrams, the graph encoder of arXiv:2606.23791 over the tree diagrams
      \item Context vector, $c_p\in\mathbb R^{16}$ per trained process added to each particle's encoding, $c_p=0$ on half the events; a new process takes $c=0$, or $c$ fitted on $K$ of its events with the network frozen
    \end{itemize}
  \end{minipage}
  \vfill
\end{frame}
