def hybrid_cp(P):
    """c_p of one process's momenta P (n, N, 4): mean over events of qperp_logs - (N - 4) log s_hat (0.0 for a 2 -> 2)."""
    pp = np.asarray(P, dtype=np.float64)
    q = pp[:, 0] + pp[:, 1]
    ls = np.log(np.abs(q[:, 0] ** 2 - (q[:, 1:] ** 2).sum(axis=1)))
    return float(np.mean(qperp_logs(pp) - (pp.shape[1] - 4) * ls))

def hybrid_cps(names, events_of, frozen=None):
    """({name: c_p}, [reused names]). A name in `frozen` keeps that value and events_of(name) is NOT called;
    any other is fitted with hybrid_cp(events_of(name))."""
    frozen = frozen or {}
    cps, reused = {}, []
    for n in names:
        if n in frozen:
            cps[n] = float(frozen[n]); reused.append(n)
        else:
            cps[n] = hybrid_cp(events_of(n))
    return cps, reused
