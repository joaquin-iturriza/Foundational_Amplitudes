ROLES = ("train", "val", "test")
def recipe_role_counts(recipe_yaml):
    rec = yaml.safe_load(open(recipe_yaml))
    return {p["name"]: {r: int(p.get(f"n_{r}", 0)) for r in ROLES} for p in rec["processes"]}
def stand_in_role(counts, process, role, scored_split):
    c = counts[process]       # own role, else the scored split, else another role, else own (0)
    for r in (role, scored_split, *ROLES):
        if c.get(r, 0) > 0:
            return r, c[r]
    return role, 0
def stand_in_split_set(real_ensure_split_set, counts, scored_split):
    def split(specs, role, seed, dest_dir=None, require_cache=False):   # real's signature
        out = {}
        for s in specs:       # spec keys: process, sqrts_min, sqrts_max, n_events (experiment.py l.1500)
            r, n = stand_in_role(counts, s["process"], role, scored_split)
            out.update(real_ensure_split_set([{**s, "n_events": n}], role=r, seed=seed,
                                             dest_dir=dest_dir, require_cache=require_cache))
        return out
    return split
