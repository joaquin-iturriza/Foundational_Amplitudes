def owned_ended(st, eid, ended, t):
    """True when every ended run (ids) belongs to a working step agent of the entry: it was submitted after the
    agent started, or finished by the agent's last transcript record."""
    ws = []
    for x in (st.get(eid) or {}).get("steps") or []:
        a = None if step_problem(st, eid, x) else subagent_activity(x.get("agent"))
        if a and not a[1] and 0 <= (t - a[0]).total_seconds() / 60 <= STEP_MIN:
            ws.append((parse(x.get("at")), a[0]))          # (agent start, last record)
    c = db()
    sub = {r: s for r, s in c.execute("SELECT id, submitted FROM runs WHERE project=?", [PROJECT])} if c else {}
    return bool(ws) and all(any((sub.get(r) and parse(sub[r]) >= s0) or parse(fin) <= last for s0, last in ws)
                            for r, fin in ended)
