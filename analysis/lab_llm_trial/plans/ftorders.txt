def parent_orders_flag(parent_dir):
    """(orders_from_diagrams flag for the fine-tune, the parent's source, how it was read). An alpha_s-only parent is refused."""
    try:
        src = json.load(open(os.path.join(parent_dir, "data_stats.json"))).get("target_physics_orders_source")
    except FileNotFoundError:
        src = None
    how = "data_stats.json target_physics_orders_source"
    if src is None:
        sys.path.insert(0, ROOT)
        from experiment import AmplitudeExperiment      # import here: heavy, only needed for old parents
        src, how = AmplitudeExperiment._infer_orders_source(parent_dir)
    if src == "table":
        return False, src, how
    if src == "diagrams_highest_alphas":
        return True, src, how
    raise ValueError(f"parent {parent_dir}: coupling-monomial source {src!r} ({how}) cannot be matched by a fine-tune "
                     "(only 'table' and 'diagrams_highest_alphas' are supported)")

def test_catalog_ft_orders_follow_parent():
    import json, tempfile
    import tools.catalog_ft as cf
    with tempfile.TemporaryDirectory() as d:
        def parent(src):
            json.dump({"target_physics_orders_source": src}, open(os.path.join(d, "data_stats.json"), "w"))
            return d
        assert cf.parent_orders_flag(parent("table"))[:2] == (False, "table")
        assert cf.parent_orders_flag(parent("diagrams_highest_alphas"))[:2] == (True, "diagrams_highest_alphas")
        try:
            cf.parent_orders_flag(parent("diagrams_alphas_only"))
            raise RuntimeError("an alpha_s-only parent was accepted")
        except ValueError:
            pass
    for flag in (True, False):
        cmd = cf.command("runs/x", "MuonAdamW", "r.yaml", 10, 176, "rd", "rd_result.json", "e", orders_from_diagrams=flag)
        assert f"++data.orders_from_diagrams={str(flag).lower()}" in cmd
    assert not any("orders_from_diagrams" in c for c in cf.command("runs/x", "MuonAdamW", "r.yaml", 10, 176, "rd", "r.json", "e"))
